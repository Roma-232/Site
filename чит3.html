Пример кода: Генератор чисел Фибоначчи

Python

def fibonacci_generator(n):
    """Генератор последовательности Фибоначчи"""
    a, b = 0, 1
    count = 0
    while count < n:
        yield a
        a, b = b, a + b
        count += 1

# Использование генератора
print("Первые 10 чисел Фибоначчи:")
for num in fibonacci_generator(10):
    print(num, end=" ")

# Использование спискового включения
fib_list = [num for num in fibonacci_generator(10)]
print(f"\nСписок: {fib_list}")

# Использование функции next()
gen = fibonacci_generator(5)
print(f"Первое число: {next(gen)}")
print(f"Второе число: {next(gen)}")
